\section{Motivation}

% ============================================================
\begin{frame}{Warum luftatmende elektrische Antriebe (ABEP)?}
  \small
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item \textbf{VLEO} (\emph{Very Low Earth Orbit}, ca.\ 150--250\,km): bessere Erdbeobachtung, mehr Datendurchsatz, weniger Strahlenbelastung
        \item Dort dominiert der \alert{Restatmosphaerenwiderstand} -- ohne Schubkompensation zerfaellt die Bahn in Tagen bis Wochen
        \item Klassisch: mitgefuehrter Treibstoff (Xenon) $\Rightarrow$ begrenzte Missionsdauer
        \item \textbf{ABEP-Idee}: Restatmosphaere (O$_2$, N$_2$) selbst als Treibstoff nutzen
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}[node distance=4mm, every node/.style={font=\scriptsize, align=center}]
        \node[draw=giessenblau, thick, rounded corners, fill=hellblau, minimum width=2.8cm, minimum height=0.65cm] (in) {Lufteinlass};
        \node[draw=giessenblau, thick, rounded corners, fill=hellblau, minimum width=2.8cm, minimum height=0.65cm, below=of in] (ion) {Ionisation (RIT)};
        \node[draw=giessenblau, thick, rounded corners, fill=hellblau, minimum width=2.8cm, minimum height=0.65cm, below=of ion] (acc) {Beschleunigung};
        \node[draw=giessenblau, thick, rounded corners, fill=hellblau, minimum width=2.8cm, minimum height=0.65cm, below=of acc] (neu) {Neutralisation};
        \draw[-{Latex}, giessenblau, thick] (in) -- (ion);
        \draw[-{Latex}, giessenblau, thick] (ion) -- (acc);
        \draw[-{Latex}, giessenblau, thick] (acc) -- (neu);
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{alertblock}{Kernidee}
    Treibstoff ist praktisch unbegrenzt vorhanden -- die Herausforderung liegt im Triebwerk, das mit reaktiven Gasen statt Xenon arbeiten muss
  \end{alertblock}
\end{frame}

\section{Technologie am Institut}

% ============================================================
\begin{frame}{RIT-Technologie und Neutralisator an der JLU}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Funktionsprinzip RIT}
      \begin{itemize}
        \item Hochfrequenz-induktive Plasmaerzeugung, \alert{elektrodenlos} in der Ionisationskammer
        \item Gitteroptik extrahiert und beschleunigt den Ionenstrahl
        \item Keine Kathode im Entladungsraum $\Rightarrow$ hohe Lebensdauer, unempfindlich gegenueber reaktiven Gasen
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Neutralisator}
      \begin{itemize}
        \item Baugleiches HF-Prinzip, aber als \alert{Elektronenquelle} betrieben (Projekt \emph{ECHION})
        \item Liefert Elektronen zur Ladungs- und Stromneutralisation des Ionenstrahls
        \item Kopplung ueber eine \alert{Plasmabruecke} zwischen Neutralisator und Strahl
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{block}{Einordnung}
    Die AG Ionentriebwerke (Prof.\ Klar) entwickelt RIT-Technologie seit ueber 50 Jahren; ABEP und die dafuer noetige HF-Neutralisatorentwicklung (ECHION, DLR-gefoerdert) sind aktuelle Schwerpunkte \cite{holste2020}
  \end{block}
\end{frame}

\section{Ausgangslage}

% ============================================================
\begin{frame}{Ausgangslage: Vorarbeiten von Zorn et al.\ (2025)}
  \small
  \begin{itemize}
    \item RIT-10 und HF-Neutralisator wurden \textbf{getrennt} mit O$_2$, N$_2$ und Mischungen charakterisiert \cite{zorn2025}
    \item Diagnostik: Langmuirsonden, optische Emissionsspektroskopie, Faraday-Cups, Retarding-Potential-Analyzer, Massenspektrometrie, Raman-Spektroskopie
  \end{itemize}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Elektronendichte: $10^{16}$--$10^{17}\,\mathrm{m}^{-3}$
        \item Elektronentemperatur: 5--8\,eV
        \item Strahldivergenz: $\approx$20$^\circ$ (N$_2$) bis 28$^\circ$ (O$_2$)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Neutralisatorstrom bis $\approx$100\,mA bei 2\,MHz, 100\,W
        \item Dissoziationsgrad O$_2$: 25--41\,\%
        \item Betriebsfenster: 6--20\,sccm, 1{,}3--2{,}0\,MHz, 1400\,V Gitterspannung
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{alertblock}{Erkenntnislücke}
    Triebwerk und Neutralisator wurden bisher \textbf{einzeln} untersucht -- das \textbf{gekoppelte} Betriebsverhalten (Plasmabruecke, Winkelabhaengigkeit, Rueckwirkung auf Bauteile) ist noch offen
  \end{alertblock}
\end{frame}

\begin{frame}{Diagnostik-Schwerpunkt: Langmuirsonde in der Plasmabrücke}
  \begin{itemize}
    \item Erfassung von Elektronendichte $n_e$, Elektronentemperatur $T_e$ und Plasmapotential entlang der Bruecke
    \item Ableitung der \alert{Kopplungsspannung} zwischen Neutralisator-Kathode und Tankmasse als Mass fuer effiziente Elektroneneinkopplung
    \item Methodik angelehnt an etablierte Sondendiagnostik an Ionenstrahl-Neutralisatoren \cite{roibas2013,appl2024}
  \end{itemize}
  \vspace{0.5em}
  \begin{alertblock}{Zu beachten}
    Sondenstoerung des Plasmas, Positioniergenauigkeit, Kalibrierung -- Details siehe Feedback
  \end{alertblock}
